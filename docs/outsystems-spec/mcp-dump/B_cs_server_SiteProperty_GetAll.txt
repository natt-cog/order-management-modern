(OutSystems.Model.IESpace eSpace) => {
    /*** creating IServerAction 'SiteProperty_GetAll' ***/
    var siteProperty_GetAll = eSpace.CreateServerAction("SiteProperty_GetAll");
    siteProperty_GetAll.Description = "Returns all site properties";
    var siteProperties = eSpace.Folders.Named("SiteProperties");
    siteProperty_GetAll.Folder = siteProperties;
    siteProperty_GetAll.Public = true;
    siteProperty_GetAll.Shareable = true;
    var startNode = siteProperty_GetAll.CreateNode<OutSystems.Model.Logic.Nodes.IStartNode>();
    var siteProperty_GetCurrency = siteProperty_GetAll.CreateNode<OutSystems.Model.Logic.Nodes.IExecuteServerActionNode>("SiteProperty_GetCurrency");
    var siteProperty_GetCurrency2 = eSpace.ServerActions.Named("SiteProperty_GetCurrency");
    siteProperty_GetCurrency.Action = siteProperty_GetCurrency2;
    startNode.Target = siteProperty_GetCurrency;
    var assignNode = siteProperty_GetAll.CreateNode<OutSystems.Model.Logic.Nodes.IAssignNode>().ConnectedBelow(siteProperty_GetCurrency, 1368);
    assignNode.Label = "Currency, DecimalSeparator, GroupSeparator";
    var assignment = assignNode.CreateAssignment("Currency", "SiteProperty_GetCurrency.Currency");
    var assignment2 = assignNode.CreateAssignment("DecimalSeparator", "SiteProperty_GetCurrency.DecimalSeparator");
    var assignment3 = assignNode.CreateAssignment("GroupSeparator", "SiteProperty_GetCurrency.GroupSeparator");
    var siteProperty_GetDashboard = siteProperty_GetAll.CreateNode<OutSystems.Model.Logic.Nodes.IExecuteServerActionNode>("SiteProperty_GetDashboard").ConnectedBelow(assignNode, 1368);
    var siteProperty_GetDashboard2 = eSpace.ServerActions.Named("SiteProperty_GetDashboard");
    siteProperty_GetDashboard.Action = siteProperty_GetDashboard2;
    var assignNode2 = siteProperty_GetAll.CreateNode<OutSystems.Model.Logic.Nodes.IAssignNode>().ConnectedBelow(siteProperty_GetDashboard, 1368);
    assignNode2.Label = "Dashboard";
    var assignment4 = assignNode2.CreateAssignment("Dashboard_PersonalGoal", "SiteProperty_GetDashboard.Dashboard_PersonalGoal");
    var assignment5 = assignNode2.CreateAssignment("Dashboard_TeamGoalPercent", "SiteProperty_GetDashboard.Dashboard_TeamGoalPercent");
    var assignment6 = assignNode2.CreateAssignment("Dashboard_TargetValue", "SiteProperty_GetDashboard.Dashboard_TargetValue");
    var endNode = siteProperty_GetAll.CreateNode<OutSystems.Model.Logic.Nodes.IEndNode>();
    assignNode2.Target = endNode;
    var currency = siteProperty_GetAll.CreateOutputParameter("Currency");
    currency.DataType = eSpace.TextType;
    currency.SetDefaultValue("\"$\"");
    currency.Description = "Currency symbol for number formatting";
    var decimalSeparator = siteProperty_GetAll.CreateOutputParameter("DecimalSeparator");
    decimalSeparator.DataType = eSpace.TextType;
    decimalSeparator.Description = "Decimal separator for number formatting";
    var groupSeparator = siteProperty_GetAll.CreateOutputParameter("GroupSeparator");
    groupSeparator.DataType = eSpace.TextType;
    groupSeparator.SetDefaultValue("\",\"");
    groupSeparator.Description = "Group separator for number formatting";
    var dashboard_PersonalGoal = siteProperty_GetAll.CreateOutputParameter("Dashboard_PersonalGoal");
    dashboard_PersonalGoal.DataType = eSpace.CurrencyType;
    dashboard_PersonalGoal.SetDefaultValue("50000");
    var dashboard_TeamGoalPercent = siteProperty_GetAll.CreateOutputParameter("Dashboard_TeamGoalPercent");
    dashboard_TeamGoalPercent.DataType = eSpace.IntegerType;
    dashboard_TeamGoalPercent.SetDefaultValue("54");
    dashboard_TeamGoalPercent.Description = "Percent of company goal for team goal";
    var dashboard_TargetValue = siteProperty_GetAll.CreateOutputParameter("Dashboard_TargetValue");
    dashboard_TargetValue.DataType = eSpace.CurrencyType;
    dashboard_TargetValue.SetDefaultValue("7500000");
    dashboard_TargetValue.Description = "Target value to be shown on dashboard";
}
